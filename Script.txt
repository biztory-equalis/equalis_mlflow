//Berekening Solvabiliteit Dossier
TempSolvabiliteit:
NOCONCATENATE LOAD DISTINCT [TDOS_ID] AS [Dossier ID],
	[TDOS_ID_HOOFD] AS [Hoofddossier ID]
FROM [$(vExportFolder)\front_end\Dossier.qvd] (qvd);
RIGHT JOIN LOAD DISTINCT [TDOS_ID] AS [Dossier ID],
	[TPAR_ID]
FROM [$(vExportFolder)\front_end\TDOSPAR.qvd] (qvd)
WHERE [Type_code]='VERW';
LEFT JOIN LOAD [TPAR_ID],
	[Solvabiliteit_Code_Financieel]
FROM [$(vExportFolder)\front_end\TPAR.qvd] (qvd);
LEFT JOIN LOAD [TPAR_ID],
	[TQSOLVABILITEIT_ID]
FROM [$(vExportFolder)\front_end\TPARSOLVABILITEIT.qvd] (qvd)
WHERE [Gebruikt]=1;
LEFT JOIN LOAD [TQSOLVABILITEIT_ID],
	1 AS [Aantal Insolvabel],
	[Solvability_dut] AS [Solvabiliteit Subtype]
FROM [$(vExportFolder)\TQSOLVABILITEIT.qvd] (qvd)
WHERE [Solvability_dut] <>'Solvabel';

SolvabiliteitDossier:
NOCONCATENATE LOAD [Hoofddossier ID],
	If(Sum([Aantal Insolvabel]) > 0,If(Count(DISTINCT [Solvabiliteit Subtype]) >=2,'Insolvabel',MaxString([Solvabiliteit Subtype])),'Solvabel') AS [Solvabiliteit Dossier],
	Max([Solvabiliteit_Code_Financieel]) AS [Dossier Solvabiliteit Code]
RESIDENT TempSolvabiliteit
GROUP BY [Hoofddossier ID];

DROP TABLE TempSolvabiliteit;

	If(Year([DatumCreatie]) >= (Year(Today())-7) AND Year([DatumCreatie]) <= Year(Today()),'Q' & Ceil(Month([DatumCreatie])/3),Null()) AS [Creatie Kwartaal],

//Factuurbedrag
TempStatusGroepering:
NOCONCATENATE LOAD [Hoofddossier ID] RESIDENT Dossiers WHERE [Berekening GDW]= 'Afgehandeld' AND [Dossier Status Code] = 'STPOPD';
LEFT JOIN LOAD [Hoofddossier ID],
	Sum([Bedrag]) AS [Dossier Bedrag Factuur]
RESIDENT Dossiers
WHERE [Berekening GDW]= 'Afgehandeld' AND [Dossier Status Code] = 'STPOPD' AND Match([Booking_code],'HSZI','HSMI','DW')
GROUP BY [Hoofddossier ID];
//Doorstorting beschikbaar
LEFT JOIN LOAD [Hoofddossier ID],
	Sum([Bedrag]) AS [Dossier Doorgestort]
RESIDENT Dossiers
WHERE [Berekening GDW]= 'Afgehandeld' AND [Dossier Status Code] = 'STPOPD' AND [Booking_code] ='DRSBES'
GROUP BY [Hoofddossier ID];
//Betaling na interventie
LEFT JOIN LOAD [Hoofddossier ID],
	Sum([Bedrag]) AS [Dossier Betaald Na Interventie]
RESIDENT Dossiers
WHERE [Berekening GDW]= 'Afgehandeld' AND [Dossier Status Code] = 'STPOPD' AND [Booking_code] ='BETBTK' AND [BetalingVoorInterventieFlag] = 0
GROUP BY [Hoofddossier ID];

StatusGroepering:
NOCONCATENATE LOAD [Hoofddossier ID],
	Sum([Dossier Bedrag Factuur]) AS [Dossier Bedrag Factuur],
	Sum([Dossier Doorgestort]) + Sum([Dossier Betaald Na Interventie]) AS [Dossier Bedrag Reeds Geind]
RESIDENT TempStatusGroepering
GROUP BY [Hoofddossier ID];
LEFT JOIN LOAD [Hoofddossier ID],
	[Solvabiliteit Dossier],
	[Vorderingsparameter]
RESIDENT Dossiers;

DROP TABLE TempStatusGroepering;

LEFT JOIN (Dossiers)
LOAD [Hoofddossier ID],
	If([Dossier Bedrag Factuur] <= [Dossier Bedrag Reeds Geind]
		,'Geregeld bij klant'
		,If([Dossier Bedrag Factuur] < [Vorderingsparameter] AND NOT IsNull([Vorderingsparameter])
			,'Bedrag te klein om te dagvaarden'
			,If([Solvabiliteit Dossier] <> 'Solvabel'
				,'Insolvabel'
				,'Gestopt door klant'
			)
		)
	) AS [Status Groepering]
RESIDENT StatusGroepering;

DROP TABLE StatusGroepering;
