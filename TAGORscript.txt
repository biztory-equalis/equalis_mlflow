///$tab sub
sub translate(vLanguage, vSource, vTarget, vKey_field, vDesc_field, vNew_desc_field)

	for vCount = 1 to SubStringCount('$(vLanguage)','|')+1
	
		let vTaal = SubField('$(vLanguage)','|',$(vCount));
		
		left join($(vTarget))
		load
			$(vKey_field),
			$(vDesc_field)			as [$(vNew_desc_field)_$(vTaal)]
		resident $(vSource)
			where mixmatch(Taal,'$(vTaal)')>0;
		
	next

	store $(vTarget) into $(vTarget).qvd;
	drop table $(vTarget),$(vSource);
end sub
